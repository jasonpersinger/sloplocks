**Goal:** Build a Linux desktop app (`~/isomount/`) for mounting and unmounting ISO files via a GTK4 window with a system tray icon.

**Architecture:** Two-layer design — `backend.py` wraps `udisksctl`/`losetup` subprocess calls with no GTK imports (fully testable), `app.py` builds the GTK4 window and tray icon, `__main__.py` handles single-instance enforcement and launches the main loop.

**Tech Stack:** Python 3.11+, GTK4 (PyGObject / `python-gobject`), `pystray` + `Pillow` (optional, for system tray — works independently of GTK version), `udisksctl` + `losetup` (system tools, no sudo required), pytest + unittest.mock.

```python
# ~/isomount/isomount/backend.py
import json
import re
import subprocess
from pathlib import Path

PROC_MOUNTS_PATH = "/proc/mounts"


def list_mounted_isos() -> list[dict]:
    """Returns list of {loop_device, iso_path, mount_point} dicts."""
    result = subprocess.run(
        ["losetup", "-l", "-J"],
        capture_output=True, text=True,
    )
    if result.returncode != 0:
        return []

    try:
        data = json.loads(result.stdout)
        loop_devices = data.get("loopdevices", [])
    except (json.JSONDecodeError, KeyError):
        return []

    iso_loops = [
        d for d in loop_devices
        if d.get("back-file", "").lower().endswith(".iso")
    ]
    if not iso_loops:
        return []

    try:
        mounts_text = Path(PROC_MOUNTS_PATH).read_text()
    except OSError:
        mounts_text = ""

    mount_map: dict[str, str] = {}
    for line in mounts_text.splitlines():
        parts = line.split()
        if len(parts) >= 2:
            mount_map[parts[0]] = parts[1]

    return [
        {
            "loop_device": d["name"],
            "iso_path": d["back-file"],
            "mount_point": mount_map[d["name"]],
        }
        for d in iso_loops
        if d["name"] in mount_map
    ]
```

```python
def mount_iso(path: str) -> tuple[str, str]:
    """Returns (loop_device, mount_point). Raises RuntimeError on failure."""
    path = str(Path(path).resolve())

    result = subprocess.run(
        ["udisksctl", "loop-setup", "-f", path],
        capture_output=True, text=True,
    )
    if result.returncode != 0:
        raise RuntimeError(result.stderr.strip() or "Failed to set up loop device")

    match = re.search(r"as (/dev/loop\d+)", result.stdout)
    if not match:
        raise RuntimeError(f"Could not parse loop device from output: {result.stdout!r}")
    loop_dev = match.group(1)

    result = subprocess.run(
        ["udisksctl", "mount", "-b", loop_dev],
        capture_output=True, text=True,
    )
    if result.returncode != 0:
        subprocess.run(["udisksctl", "loop-delete", "-b", loop_dev], capture_output=True)
        raise RuntimeError(result.stderr.strip() or "Failed to mount")

    match = re.search(r"at (.+?)\.", result.stdout)
    if not match:
        raise RuntimeError(f"Could not parse mount point from output: {result.stdout!r}")

    return loop_dev, match.group(1).strip()
```

```python
def unmount_iso(loop_device: str) -> None:
    """Raises RuntimeError on failure."""
    result = subprocess.run(
        ["udisksctl", "unmount", "-b", loop_device],
        capture_output=True, text=True,
    )
    if result.returncode != 0:
        raise RuntimeError(result.stderr.strip() or "Failed to unmount")

    result = subprocess.run(
        ["udisksctl", "loop-delete", "-b", loop_device],
        capture_output=True, text=True,
    )
    if result.returncode != 0:
        raise RuntimeError(result.stderr.strip() or "Failed to delete loop device")
```

```python
def _on_browse(self, _button):
    dialog = Gtk.FileChooserDialog(
        title="Select ISO File",
        transient_for=self,
        action=Gtk.FileChooserAction.OPEN,
    )
    dialog.add_button("_Cancel", Gtk.ResponseType.CANCEL)
    dialog.add_button("_Open", Gtk.ResponseType.ACCEPT)

    iso_filter = Gtk.FileFilter()
    iso_filter.set_name("ISO images")
    iso_filter.add_pattern("*.iso")
    iso_filter.add_pattern("*.ISO")
    dialog.add_filter(iso_filter)

    dialog.connect("response", self._on_file_chosen)
    dialog.present()

def _on_file_chosen(self, dialog, response):
    if response == Gtk.ResponseType.ACCEPT:
        f = dialog.get_file()
        if f:
            self.path_entry.set_text(f.get_path())
    dialog.destroy()

def _on_mount(self, _button):
    path = self.path_entry.get_text().strip()
    if not path:
        self._show_error("No file selected", "Use Browse to select an ISO file first.")
        return

    from isomount import backend
    try:
        loop_dev, mount_point = backend.mount_iso(path)
    except RuntimeError as e:
        self._show_error("Mount failed", str(e))
        return

    self._add_row(loop_dev, path, mount_point)
    self.path_entry.set_text("")

def _show_error(self, title: str, message: str):
    dialog = Gtk.MessageDialog(
        transient_for=self,
        modal=True,
        message_type=Gtk.MessageType.ERROR,
        buttons=Gtk.ButtonsType.OK,
        text=title,
    )
    dialog.format_secondary_text(message)
    dialog.connect("response", lambda d, _: d.destroy())
    dialog.present()

def _add_row(self, loop_device: str, iso_path: str, mount_point: str):
    pass  # implemented in Task 7
```

```python
        "iso_path": iso_path,
        "mount_point": mount_point,
        "row_widget": row,
    })
    self._update_status()

def _open_in_filemanager(self, mount_point: str):
    import subprocess
    try:
        subprocess.Popen(["xdg-open", mount_point])
    except FileNotFoundError:
        self._show_error("Cannot open", "xdg-open not found on this system.")

def _on_unmount(self, loop_device: str, row: Gtk.ListBoxRow):
    from isomount import backend
    try:
        backend.unmount_iso(loop_device)
    except RuntimeError as e:
        self._show_error("Unmount failed", str(e))
        return
    self._remove_row(row, loop_device)

def _remove_row(self, row: Gtk.ListBoxRow, loop_device: str):
    self.list_box.remove(row)
    self._iso_rows = [r for r in self._iso_rows if r["loop_device"] != loop_device]
    self._update_status()

def _populate_from_backend(self):
    """Called on startup — pre-populate list with already-mounted ISOs."""
    from isomount import backend
    for entry in backend.list_mounted_isos():
        self._add_row(entry["loop_device"], entry["iso_path"], entry["mount_point"])
```

## Task 8: GUI — Tray Icon

**Files:**
- Modify: `~/isomount/isomount/app.py`

Use `pystray` for the tray icon — it runs in its own thread independently of GTK, avoiding the GTK3/GTK4 menu incompatibility. If `pystray` import fails, the app still runs but close = quit.

- [ ] **Step 1: Add tray setup to IsoMountApp**

Add at the top of `app.py`, after existing imports:

```python

try:
    import pystray
    from PIL import Image as PilImage
    _HAS_TRAY = True
except ImportError:
    _HAS_TRAY = False
```

Replace `do_activate` in `IsoMountApp` and add tray helpers:

```python
def do_activate(self):
    if self.window is None:
        self.window = IsoMountWindow(self)
        self._tray_icon = None
        if _HAS_TRAY:
            self._start_tray()
            self.window.connect("close-request", self._on_close_request)
    self.window.present()

def _start_tray(self):
    icon_path = RESOURCES / "icon.svg"
    # pystray needs a PIL Image; convert SVG → PNG in memory via cairosvg if
    # available, otherwise fall back to a simple generated image.
    # Draw a simple disc icon with PIL (no extra deps needed)
    img = PilImage.new("RGBA", (48, 48), (0, 0, 0, 0))
    from PIL import ImageDraw
    draw = ImageDraw.Draw(img)
    draw.ellipse([4, 4, 44, 44], fill=(51, 51, 51, 255), outline=(57, 255, 20, 255), width=2)
    draw.ellipse([20, 20, 28, 28], fill=(57, 255, 20, 255))

    menu = pystray.Menu(
        pystray.MenuItem("Show Window", self._tray_show),
        pystray.Menu.SEPARATOR,
        pystray.MenuItem("Quit", self._tray_quit),
    )
    self._tray_icon = pystray.Icon("isomount", img, "ISOMount", menu)
    t = threading.Thread(target=self._tray_icon.run, daemon=True)
    t.start()

def _tray_show(self, _icon, _item):
    GLib.idle_add(self.window.present)

def _tray_quit(self, _icon, _item):
    if self._tray_icon:
        self._tray_icon.stop()
    GLib.idle_add(self.quit)

def _on_close_request(self, _window):
    self.window.hide()
    return True  # prevent default destroy
```

**Note:** The tray icon is drawn programmatically with PIL — no SVG conversion needed.
